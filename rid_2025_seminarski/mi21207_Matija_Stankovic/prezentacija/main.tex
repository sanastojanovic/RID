\section{Uvod}

\subsection{Veštačka inteligencija - nekada i danas}

\begin{frame}{\frametitlesection}

\begin{itemize}
    \item Veštačka inteligencija nekad: \textbf{mašine koje imitiraju ljudsko razmišljanje}
    \begin{itemize}
        \item Ograničena primena i mali domet
    \end{itemize}
    \item Veštačka inteligencija danas: \textbf{mašinsko učenje i obrada velikih količina podataka}
    \begin{itemize}
    \item Prisutna u svakodnevnom životu (pretraživači, društvene mreže, obrazovanje, poslovanje...)
    \end{itemize}
    \item Prelomna tačka razvoja: \textbf{LLM modeli}
\end{itemize}

\end{frame}

\section{LLM modeli i ChatGPT}

\subsection{Pregled tehnologije}

\begin{frame}{\frametitlesection}

\begin{itemize}
    \item Razvoj LLM modela zasnovan je na \textbf{dubokom učenju}
    \begin{itemize}
        \item Značajan napredak u \textbf{obradi prirodnog jezika}
        \item Trenirani na velikim količinama tekstualnih podataka
    \end{itemize}
    \item \textbf{Transformer arhitektura}
    \begin{itemize}
        \item Obrada čitavih sekvenci teksta odjednom
        \item Uočavanje odnosa između reči u različitim delovima teksta
    \end{itemize}
    \item Razlike u odnosu na ranije sisteme:
    \begin{itemize}
            \item Nisu zasnovani na ručno definisanim pravilima
            \item Poseduju sposobnost \textbf{generalizacije}
    \end{itemize}
\end{itemize}

\end{frame}

\subsection{ChatGPT i nedostaci LLM modela}

\begin{frame}{\frametitlesection}

\begin{itemize}
    \item \textbf{ChatGPT} (OpenAI)
    \begin{itemize}
        \item Zasnovan na GPT arhitekturi
        \item Komunikacija sa korisnicima putem prirodnog jezika
        \item Prilagođavanje odgovora toku razgovora
        \item Široka primena
    \end{itemize}
    \item Ključna prednost: \textbf{visoka dostupnost i jednostavnost korišćenja}
    \item \textbf{Ograničenja}
    \begin{itemize}
        \item Nepouzdanost i halucinacije
        \item Odsustvo stvarnog razumevanja
        \item Pristrasnost u podacima
        \item Zavisnost korisnika od modela
\end{itemize}
\end{itemize}

\end{frame}

\section{ChatGPT kao izvor informacija}
\subsection{Prednosti i rizici}

\begin{frame}{\frametitlesection}

\begin{itemize}
    \item \textbf{Prednosti}
    \begin{itemize}
        \item Laka dostupnost
        \item Brz pristup informacijama
        \item Prilagođenost pitanjima i nivou znanja korisnika
        \item Koristan u samostalnom i neformalnom učenju
    \end{itemize}
    \item \textbf{Rizici nekritičke upotrebe}
    \begin{itemize}
        \item Nekritičko prihvatanje odgovora
        \item Površno usvajanje znanja
        \item Privatnosni i bezbednosni problemi
    \end{itemize}
\end{itemize}

\end{frame}


\subsection{Problem halucinacija i promena načina učenja i pretrage}

\begin{frame}{\frametitlesection}

\begin{itemize}
    \item \textbf{Halucinacije}
    \begin{itemize}
        \item Uverljivi, ali netačni odgovori
        \item Izmišljanje činjenica, izvora ili događaja
        \item Nemogućnost provere izvora u realnom vremenu
    \end{itemize}

    \item ChatGPT nije autoritativni izvor znanja: \textbf{neophodna dodatna provera informacija}
    \item \textbf{Promena načina učenja i pretrage}
    \begin{itemize}
        \item Direktna pitanja umesto klasične pretrage
        \item Značaj formulacije upita (prompt-a)
    \end{itemize}
\end{itemize}
\end{frame}

\section{Uticaj ChatGPT-a na opšte znanje}
\subsection{Promena uloge opšteg znanja}

\begin{frame}{\frametitlesection}
\begin{itemize}
    \item \textbf{Opšte znanje nekada}
    \begin{itemize}
        \item Skup trajno zapamćenih informacija 
        \item Sticano kroz formalno obrazovanje, knjige, medije i lično iskustvo
    \end{itemize}
    \item Sa pojavom ChatGPT-a:
    \begin{itemize}
        \item \textbf{Znanje „na zahtev“}
        \item Fokus se pomera sa \textbf{šta znamo} na \textbf{kako postavljamo pitanje}
    \end{itemize}
    \item Opšte znanje:
    \begin{itemize}
        \item veština snalaženja u informacijama
    \end{itemize}
\end{itemize}
\end{frame}


\subsection{Percepcija znanja i akademski kontekst}

\begin{frame}{\frametitlesection}
\begin{itemize}
    \item ChatGPT omogućava:
    \begin{itemize}
        \item Pojednostavljivanje složenih tema
        \item Brz i pristupačan \textbf{osnovni uvid}
        \item Niži prag za ulazak u novu oblast
        \item Veću informisanost korisnika
    \end{itemize}
    \item Rizik:
    \begin{itemize}
        \item \textbf{Iluzija razumevanja}: izostavljanje složenijih aspekata problema
        \item Površno znanje bez dubljeg uvida
    \end{itemize}
    \item U akademskom kontekstu:
    \begin{itemize}
        \item \textbf{Veća odgovornost}
        \item Ozbiljnije posledice nepravilne upotrebe 
    \end{itemize}
\end{itemize}
\end{frame}

\section{ChatGPT u obrazovanju}

\subsection{Primena u obrazovanju}

\begin{frame}{\frametitlesection}
\begin{itemize}
    \item Sve češća primena u visokom obrazovanju 
    \begin{itemize}
        \item Objašnjavanje složenih pojmova
        \item Generisanje primera
        \item Pomoć pri pisanju i strukturiranju tekstova
    \end{itemize}
    \item \textbf{Personalizacija učenja} 
    \begin{itemize}
        \item Prilagođavanje nivou znanja studenta
        \item Fleksibilan tempo učenja
    \end{itemize}
    \item Važno ograničenje:
    \begin{itemize}
        \item Nije zamena za nastavnika i samostalno razmišljanje
    \end{itemize}
\end{itemize}
\end{frame}


\subsection{Iskustva i izazovi}

\begin{frame}{\frametitlesection}
\begin{itemize}
    \item Studenti brzo prihvataju alat 
    \item Najčešće se koristi za:
    \begin{itemize}
        \item Razumevanje gradiva
        \item Proveru razumevanja
        \item Generisanje ideja za dalji rad
    \end{itemize}
    \item Ređe se koristi kao konačni izvor informacija
    \item \textbf{Uočeni problemi:}
    \begin{itemize}
        \item Precenjivanje tačnosti odgovora
        \item Izostanak provere informacija
        \item Rešavanje zadataka bez razumevanja postupka
    \end{itemize}
\end{itemize}
\end{frame}


\subsection{Etika i mogućnosti buduće primene}

\begin{frame}{\frametitlesection}
\begin{itemize}
    \item Zloupotreba i plagijat
    \item Nejasna granica između pomoći i prepisivanja
    \item \textbf{Akademski integritet}
    \begin{itemize}
        \item Samostalnost u radu
        \item Poštovanje autorstva
        \item Transparentno korišćenje izvora
    \end{itemize}
    \item Rešenje:
    \begin{itemize}
        \item Jasna pravila upotrebe
        \item \textbf{Edukacija o odgovornom korišćenju}
        \item Razvoj namenskih AI alata za obrazovanje 
    \end{itemize}
\end{itemize}
\end{frame}
